// Kişisel ve Güvenli AI Asistanı - script.js document.addEventListener("DOMContentLoaded", () => { const sendButton = document.getElementById("send-button"); const userInput = document.getElementById("user-input"); const chatBox = document.getElementById("chat-box"); const API_URL = "https://openrouter.ai/api/v1/chat/completions"; // API Anahtarını sadece senin cihazının hafızasından alır function getPersonalApiKey() { let key = localStorage.getItem("MY_OPENROUTER_KEY"); if (!key) { key = prompt("Sisteme sadece sen erişebilirsin. Lütfen OpenRouter API Anahtarını girin:"); if (key && key.trim() !== "") { localStorage.setItem("MY_OPENROUTER_KEY", key.trim()); } else { alert("API anahtarı girilmediği için sistem çalışmayacaktır."); return null; } } return key; } async function handleSendMessage() { const promptText = userInput.value.trim(); if (!promptText) return; const apiKey = getPersonalApiKey(); if (!apiKey) return; // Kullanıcı mesajını ekrana yaz appendMessage("Sen", promptText); userInput.value = ""; const loadingId = appendMessage("Qwen AI", "Düşünüyor...", true); try { const response = await fetch(API_URL, { method: "POST", headers: { "Authorization": `Bearer ${apiKey}`, "Content-Type": "application/json", "HTTP-Referer": window.location.origin, "X-Title": "Personal AI Assistant" }, body: JSON.stringify({ model: "qwen/qwen-2.5-72b-instruct", // Kullanmak istediğin model messages: [{ role: "user", content: promptText }] }) }); if (!response.ok) { if (response.status === 401) { localStorage.removeItem("MY_OPENROUTER_KEY"); // Hatalı anahtar girildiyse sıfırla throw new Error("API Anahtarı geçersiz. Sayfayı yenileyip tekrar girin."); } throw new Error(`Sunucu Hatası: ${response.status}`); } const data = await response.json(); removeMessage(loadingId); appendMessage("Qwen AI", data.choices[0].message.content); } catch (error) { console.error(error); removeMessage(loadingId); appendMessage("Sistem Hata", error.message || "Bağlantı kurulamadı."); } } function appendMessage(sender, text, isLoading = false) { if (!chatBox) return; const messageDiv = document.createElement("div"); const messageId = "msg-" + Date.now(); messageDiv.id = messageId; messageDiv.style.margin = "10px 0"; messageDiv.style.padding = "8px 12px"; messageDiv.style.borderRadius = "5px"; messageDiv.style.backgroundColor = sender === "Sen" ? "#e3f2fd" : "#f5f5f5"; messageDiv.innerHTML = `${sender}: ${text}`; chatBox.appendChild(messageDiv); chatBox.scrollTop = chatBox.scrollHeight; return messageId; } function removeMessage(id) { const element = document.getElementById(id); if (element) element.remove(); } if (sendButton && userInput) { sendButton.addEventListener("click", handleSendMessage); userInput.addEventListener("keypress", (e) => { if (e.key === "Enter") handleSendMessage(); }); } });
